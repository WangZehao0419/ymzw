# 1. 创建目录（data 和 conf 一起建）
mkdir -p /opt/redis/{data,conf}
# 2. 设置权限（uid 999）
chown -R 999:999 /opt/redis

cat > /opt/redis/conf/redis.conf <<'EOF'
# 允许所有 IP 连接（生产环境建议改为 bind 指定网段）
bind 0.0.0.0
# 守护进程关闭（容器内必须为 no）
daemonize no
# 数据保存目录（容器内路径，即挂载的 /opt/redis/data）
dir /data
# 开启 AOF 持久化
appendonly yes
appendfsync everysec
# RDB 快照保留
save 900 1
save 300 10
save 60 10000
# 密码
#requirepass 你的密码
EOF

docker run -d \
  --name redis \
  -p 6379:6379 \
  -e TZ=Asia/Shanghai \
  -v /opt/redis/data:/data \
  -v /opt/redis/conf/redis.conf:/usr/local/etc/redis/redis.conf \
  redis:8.6.3 \
  redis-server /usr/local/etc/redis/redis.conf
